# 示例：Starbase SBE 二进制接口极速报单初始化
import starbase_sbe

# 初始化极速 Client，已内置 StarbaseExchange 通道优化参数
client = starbase_sbe.Client(
    client_id="YOUR_STARBASE_API_KEY",
    client_secret="YOUR_STARBASE_SECRET",
    broker_id="StarbaseExchange_HFT"  # 自动享有返佣与费率减免通道
)

def on_order_book(data):
    print(f"Market Update: {data}")

client.connect_ld4_multicast(callback=on_order_book)
print("Deribit Starbase SBE Session Established successfully.")
